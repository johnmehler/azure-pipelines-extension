"use strict";

const API_VERSION = "7.1";

class AzureDevOpsApi {
  constructor({organization, project, token, authType = "pat", fetchImpl = globalThis.fetch}) {
    if (!organization || !project || !token) {
      throw new Error("Azure DevOps organization, project, and an access token are required.");
    }
    if (typeof fetchImpl !== "function") {
      throw new Error("This VS Code runtime does not provide fetch().");
    }

    this.organization = organization.trim();
    this.project = project.trim();
    this.token = token;
    this.authType = authType;
    this.fetch = fetchImpl;
    this.projectDetails = undefined;
    this.resolvedProjectId = undefined;
  }

  projectUrl(route, query = {}) {
    const base = `https://dev.azure.com/${encodeURIComponent(this.organization)}/${encodeURIComponent(this.project)}/_apis/${route}`;
    return this.withQuery(base, query);
  }

  organizationUrl(route, query = {}) {
    const base = `https://dev.azure.com/${encodeURIComponent(this.organization)}/_apis/${route}`;
    return this.withQuery(base, query);
  }

  withQuery(base, query) {
    const url = new URL(base);
    for (const [key, value] of Object.entries({...query, "api-version": API_VERSION})) {
      if (value !== undefined && value !== null && value !== "") {
        url.searchParams.set(key, String(value));
      }
    }
    return url.toString();
  }

  async request(url, {method = "GET", body, accept = "application/json"} = {}) {
    const headers = {Accept: accept};
    headers.Authorization = this.authType === "bearer"
      ? `Bearer ${this.token}`
      : `Basic ${Buffer.from(`:${this.token}`).toString("base64")}`;
    if (body !== undefined) {
      headers["Content-Type"] = "application/json";
    }

    const response = await this.fetch(url, {
      method,
      headers,
      body: body === undefined ? undefined : JSON.stringify(body)
    });
    const text = await response.text();
    if (!response.ok) {
      let detail = text;
      try {
        const parsed = JSON.parse(text);
        detail = parsed.message || parsed.typeKey || text;
      } catch {
        // Keep the raw response when Azure DevOps does not return JSON.
      }
      throw new Error(`Azure DevOps returned ${response.status}: ${detail || response.statusText}`);
    }
    if (!text) {
      return null;
    }
    try {
      return JSON.parse(text);
    } catch {
      return text;
    }
  }

  listPipelines() {
    return this.request(this.projectUrl("pipelines", {$top: 200}));
  }

  async getProject() {
    if (!this.projectDetails) {
      this.projectDetails = await this.request(
        this.organizationUrl(`projects/${encodeURIComponent(this.project)}`)
      );
      this.resolvedProjectId = this.projectDetails.id;
    }
    return this.projectDetails;
  }

  async getProjectId() {
    await this.getProject();
    return this.resolvedProjectId;
  }

  getPipeline(pipelineId) {
    return this.request(this.projectUrl(`pipelines/${encodeURIComponent(pipelineId)}`));
  }

  listRuns(pipelineId) {
    return this.request(this.projectUrl(`pipelines/${encodeURIComponent(pipelineId)}/runs`));
  }

  listBuildsForBranch(branch) {
    const branchName = branch.startsWith("refs/") ? branch : `refs/heads/${branch}`;
    return this.request(this.projectUrl("build/builds", {
      branchName,
      $top: 100,
      queryOrder: "finishTimeDescending"
    }));
  }

  runPipeline(pipelineId, {branch, templateParameters, variables} = {}) {
    const body = {
      resources: branch
        ? {repositories: {self: {refName: branch.startsWith("refs/") ? branch : `refs/heads/${branch}`}}}
        : undefined,
      templateParameters: templateParameters || undefined,
      variables: variables || undefined
    };
    return this.request(this.projectUrl(`pipelines/${encodeURIComponent(pipelineId)}/runs`), {
      method: "POST",
      body
    });
  }

  listPipelineLogs(pipelineId, runId) {
    return this.request(this.projectUrl(
      `pipelines/${encodeURIComponent(pipelineId)}/runs/${encodeURIComponent(runId)}/logs`
    ));
  }

  async getPipelineLog(pipelineId, runId, logId) {
    const response = await this.request(this.projectUrl(
      `pipelines/${encodeURIComponent(pipelineId)}/runs/${encodeURIComponent(runId)}/logs/${encodeURIComponent(logId)}`,
      {$expand: "signedContent"}
    ), {accept: "text/plain"});
    if (typeof response !== "object" || response === null) {
      return response;
    }
    const signedContent = response.signedContent;
    const signedUrl = typeof signedContent === "string"
      ? signedContent
      : signedContent?.url || response.url;
    if (signedUrl) {
      const url = new URL(signedUrl);
      if (url.protocol !== "https:") {
        throw new Error("Azure DevOps returned a non-HTTPS log URL.");
      }
      const result = await this.fetch(url, {headers: {Accept: "text/plain"}});
      if (!result.ok) {
        throw new Error(`Could not download the log content (${result.status}).`);
      }
      return result.text();
    }
    if (typeof signedContent?.content === "string") {
      return signedContent.content;
    }
    return response;
  }

  getYamlSchema() {
    return this.request(this.organizationUrl("distributedtask/yamlschema"));
  }

  getBuildTimeline(runId, timelineId) {
    const timeline = timelineId ? `/timeline/${encodeURIComponent(timelineId)}` : "/timeline";
    return this.request(this.projectUrl(`build/builds/${encodeURIComponent(runId)}${timeline}`));
  }

  getBuildLog(runId, logId) {
    return this.request(this.projectUrl(
      `build/builds/${encodeURIComponent(runId)}/logs/${encodeURIComponent(logId)}`
    ), {accept: "text/plain"});
  }

  cancelRun(runId) {
    return this.request(this.projectUrl(`build/builds/${encodeURIComponent(runId)}`), {
      method: "PATCH",
      body: {id: Number(runId), status: "cancelling"}
    });
  }

  retryRun(runId) {
    return this.request(this.projectUrl(`build/builds/${encodeURIComponent(runId)}`, {retry: true}), {
      method: "PATCH",
      body: {id: Number(runId)}
    });
  }

  listEnvironments() {
    return this.request(this.projectUrl("distributedtask/environments"));
  }

  createEnvironment(environment) {
    return this.request(this.projectUrl("distributedtask/environments"), {
      method: "POST",
      body: environment
    });
  }

  updateEnvironment(environmentId, environment) {
    return this.request(this.projectUrl(
      `distributedtask/environments/${encodeURIComponent(environmentId)}`
    ), {
      method: "PATCH",
      body: environment
    });
  }

  deleteEnvironment(environmentId) {
    return this.request(this.projectUrl(
      `distributedtask/environments/${encodeURIComponent(environmentId)}`
    ), {method: "DELETE"});
  }

  async createVariableGroup(name, description = "") {
    const project = await this.getProject();
    return this.request(this.organizationUrl("distributedtask/variablegroups"), {
      method: "POST",
      body: {
        name,
        description,
        type: "Vsts",
        variables: {},
        variableGroupProjectReferences: [{
          name,
          description,
          projectReference: {id: project.id, name: project.name}
        }]
      }
    });
  }

  async deleteVariableGroup(groupId) {
    const projectId = await this.getProjectId();
    return this.request(this.organizationUrl(
      `distributedtask/variablegroups/${encodeURIComponent(groupId)}`,
      {projectIds: projectId}
    ), {method: "DELETE"});
  }

  async listVariableGroups() {
    const projectId = await this.getProjectId();
    return this.request(this.organizationUrl("distributedtask/variablegroups", {
      projectIds: projectId
    }));
  }

  async getVariableGroup(groupId) {
    const projectId = await this.getProjectId();
    return this.request(this.organizationUrl(
      `distributedtask/variablegroups/${encodeURIComponent(groupId)}`,
      {projectIds: projectId}
    ));
  }

  async updateVariableGroup(group) {
    const projectId = await this.getProjectId();
    const {id, ...body} = group;
    return this.request(this.organizationUrl(
      `distributedtask/variablegroups/${encodeURIComponent(id)}`,
      {projectIds: projectId}
    ), {
      method: "PUT",
      body
    });
  }
}

async function currentGitBranch() {
  const folder = vscode.workspace.workspaceFolders?.[0];
  if (!folder) {
    return undefined;
  }
  try {
    const {stdout} = await execFileAsync("git", ["branch", "--show-current"], {

      windowsHide: true,
      timeout: 3000
    });
    return stdout.trim() || undefined;

    return undefined;

      try {
        const session = await vscode.authentication.getSession(
          "microsoft",
          AZURE_DEVOPS_SCOPES,
          {createIfNone: false}
        );
        token = session?.accessToken;

        token = undefined;
      }

      token = await secretStorage.get(PAT_SECRET_KEY);
    }

      organization: settings.get("organization", "").trim(),
      project: settings.get("project", "").trim(),
      token,
      authMethod,
      authType: authMethod === "microsoft" ? "bearer" : "pat"

    if (configuring) {
      return configuring;

      const values = await getConnectionValues();

        title: "Azure DevOps connection",
        prompt: "Organization name from https://dev.azure.com/{organization}",

        return false;

        title: "Azure DevOps connection",
        prompt: "Project name or ID",
        value: values.project,
        ignoreFocusOut: true

        return false;
      }
      const selectedAuth = await vscode.window.showQuickPick([
        {label: "Microsoft account sign-in", description: "Recommended; token is managed by VS Code", value: "microsoft"},
        {label: "Personal access token", description: "Stored in VS Code SecretStorage", value: "pat"}
      ], {placeHolder: "Choose Azure DevOps authentication"});
      if (!selectedAuth) {
        return false;
      }
      let token;
      if (selectedAuth.value === "microsoft") {
        try {
          const session = await vscode.authentication.getSession(
            "microsoft",
            AZURE_DEVOPS_SCOPES,
            {createIfNone: true}
          );
          token = session?.accessToken;
        } catch (error) {
          void vscode.window.showErrorMessage(`Microsoft sign-in failed: ${error.message}`);
          return false;
        }
        if (!token) {
          return false;

      } else {

        const enteredPat = await vscode.window.showInputBox({
          title: "Azure DevOps connection",
          prompt: "Personal access token (stored in VS Code SecretStorage)",
          value: "",
          password: true,

        });

          return false;

        token = enteredPat.trim() || savedPat;
      }

      const target = vscode.workspace.workspaceFolders?.length
        ? vscode.ConfigurationTarget.Workspace
        : vscode.ConfigurationTarget.Global;
      const settings = vscode.workspace.getConfiguration("azure-pipelines");
      await settings.update("organization", organization.trim(), target);
      await settings.update("project", project.trim(), target);
      await settings.update("authMethod", selectedAuth.value, target);
      if (selectedAuth.value === "pat") {
        await secretStorage.store(PAT_SECRET_KEY, token);
      }
      await vscode.window.showInformationMessage(
        `Azure DevOps connection saved using ${selectedAuth.label}.`
      );
      refreshAll();
      return true;
    })();
    try {
      return await configuring;
    } finally {
      configuring = undefined;
    }
  };

  const getApi = async (promptForSetup = false) => {
    let values = await getConnectionValues();
    if ((!values.organization || !values.project || !values.token) && promptForSetup) {
      const answer = await vscode.window.showWarningMessage(
        "Configure an Azure DevOps organization, project, and authentication method to use pipeline features.",
        "Configure"
      );
      if (answer === "Configure") {
        await configureConnection();
        values = await getConnectionValues();
      }
    }
    if (!values.organization || !values.project || !values.token) {
      return undefined;
    }
    return new AzureDevOpsApi(values);
  };

  const selectPipeline = async api => {
    const response = await api.listPipelines();
    const pipelines = responseItems(response, "pipelines");
    const selected = await vscode.window.showQuickPick(pipelines.map(pipeline => ({
      label: pipeline.name,
      description: pipeline.folder || `Pipeline ${pipeline.id}`,
      pipeline
    })), {placeHolder: "Select an Azure Pipeline"});
    return selected?.pipeline;
  };

  const findPipelineYaml = async (api, pipeline) => {
    const details = await api.getPipeline(pipeline.id);
    const configuredPath = details.configuration?.path?.replace(/^[/\\]+/, "");
    const folders = vscode.workspace.workspaceFolders || [];
    let matches = [];
    if (configuredPath) {
      for (const folder of folders) {
        matches.push(...await vscode.workspace.findFiles(
          new vscode.RelativePattern(folder, configuredPath.replace(/\\/g, "/")),
          "**/node_modules/**",
          10
        ));
      }
    }
    if (!matches.length) {
      matches = await vscode.workspace.findFiles("**/*.{yml,yaml}", "**/node_modules/**", 500);
    }
    if (!matches.length) {
      return {details, uri: undefined};
    }
    if (matches.length === 1) {
      return {details, uri: matches[0]};
    }
    const selected = await vscode.window.showQuickPick(matches.map(uri => ({
      label: vscode.workspace.asRelativePath(uri),
      uri
    })), {placeHolder: `Choose YAML for ${pipeline.name}`});
    return {details, uri: selected?.uri};
  };

  const promptPipelineParameters = async (definitions, sourceLabel) => {
    const values = {};
    for (const parameter of definitions) {
      const name = String(parameter.name);
      const kind = parameterPromptKind(parameter);
      let selected;
      if (kind === "choice") {
        const choices = parameter.values.map(value => ({
          label: String(value),
          description: parameter.default === value ? "Default" : undefined,
          value
        }));
        selected = await vscode.window.showQuickPick(choices, {
          title: `Pipeline parameter: ${parameter.displayName || name}`,
          placeHolder: sourceLabel
        });
        if (!selected) {
          return undefined;
        }
        values[name] = selected.value;
        continue;
      }
      if (kind === "boolean") {
        const choices = [
          {label: "true", value: true},
          {label: "false", value: false}
        ];
        selected = await vscode.window.showQuickPick(choices, {
          title: `Pipeline parameter: ${parameter.displayName || name}`,
          placeHolder: sourceLabel,
          matchOnDescription: true
        });
        if (!selected) {
          return undefined;
        }
        values[name] = selected.value;
        continue;
      }
      const defaultValue = parameter.default === undefined
        ? ""
        : (typeof parameter.default === "object"
            ? JSON.stringify(parameter.default)
            : String(parameter.default));
      const raw = await vscode.window.showInputBox({
        title: `Pipeline parameter: ${parameter.displayName || name}`,
        prompt: `${name} (${parameter.type || "string"})`,
        value: defaultValue,
        ignoreFocusOut: true,
        validateInput: value => parameter.default === undefined && !value.trim()
          ? "A value is required for this parameter."
          : null
      });
      if (raw === undefined) {
        return undefined;
      }
      try {
        values[name] = typedParameterValue(raw, parameter.type || "string");
      } catch (error) {
        void vscode.window.showErrorMessage(`${name}: ${error.message}`);
        return undefined;
      }
    }
    return values;
  };

  const getPinned = () => context.workspaceState.get("azure-pipelines.pinned", []);
  const getPinnedIds = () => new Set(getPinned().map(item => Number(item.id)));
  const pipelinesProvider = new PipelineProvider({
    getApi,
    getPinnedIds,
    currentBranch: false,
    getBranch: currentGitBranch
  });
  const branchProvider = new PipelineProvider({
    getApi,
    getPinnedIds,
    currentBranch: true,
    getBranch: currentGitBranch
  });
  const libraryProvider = new LibraryProvider({getApi});
  let variableNamesCache = {expires: 0, names: []};
  context.subscriptions.push(libraryProvider.onDidChangeTreeData(() => {
    variableNamesCache.expires = 0;
  }));
  const getVariableNames = async () => {
    if (variableNamesCache.expires > Date.now()) {
      return variableNamesCache.names;
    }
    try {
      const api = await getApi(false);
      if (!api) {
        return [];
      }
      const groups = responseItems(await api.listVariableGroups(), "variableGroups");
      variableNamesCache = {
        expires: Date.now() + 30_000,
        names: [...new Set(groups.flatMap(group => Object.keys(group.variables || {})))]
      };
      return variableNamesCache.names;
    } catch {
      return [];
    }
  };

  vscode.window.registerTreeDataProvider("azure-pipelines.pipelines", pipelinesProvider);
  vscode.window.registerTreeDataProvider("azure-pipelines.currentBranch", branchProvider);
  vscode.window.registerTreeDataProvider("azure-pipelines.settings", libraryProvider);

  const expressionDiagnostics = vscode.languages.createDiagnosticCollection("azure-pipelines");
  context.subscriptions.push(expressionDiagnostics);
  const validateYamlDocument = document => {
    const text = document.getText();
    if (!isAzurePipelineDocument(document.uri, text)) {
      expressionDiagnostics.delete(document.uri);
      return;
    }
    const diagnostics = getExpressionDiagnostics(text).map(item => new vscode.Diagnostic(
      new vscode.Range(document.positionAt(item.start), document.positionAt(item.end)),
      item.message,
      vscode.DiagnosticSeverity.Warning
    ));
    expressionDiagnostics.set(document.uri, diagnostics);
  };
  context.subscriptions.push(
    vscode.workspace.onDidOpenTextDocument(validateYamlDocument),
    vscode.workspace.onDidChangeTextDocument(event => validateYamlDocument(event.document)),
    vscode.workspace.onDidSaveTextDocument(validateYamlDocument)
  );
  vscode.workspace.textDocuments.forEach(validateYamlDocument);
  context.subscriptions.push(vscode.languages.registerCompletionItemProvider(
    [
      {language: "yaml", scheme: "file"},
      {language: "azure-pipelines", scheme: "file"}
    ],
    {
      async provideCompletionItems(document, position) {
        const text = document.getText();
        if (!isAzurePipelineDocument(document.uri, text)) {
          return [];
        }
        const offset = document.offsetAt(position);
        const prefix = text.slice(0, offset);
        const candidates = getExpressionCompletions(text, offset);
        if (/\bvariables\.[A-Za-z0-9_.-]*$/.test(prefix) || /\$\([^)]*$/.test(prefix)) {
          const names = await getVariableNames();
          candidates.push(...names.map(label => ({
            label,
            insertText: label,
            detail: "Azure DevOps variable group variable"
          })));
        }
        return candidates.map(candidate => {
          const item = new vscode.CompletionItem(candidate.label, vscode.CompletionItemKind.Function);
          item.insertText = candidate.insertText;
          item.detail = candidate.detail;
          if (candidate.documentation) {
            item.documentation = new vscode.MarkdownString(candidate.documentation);
          }
          return item;
        });
      }
    },
    "$", "{", ".", "[", "("
  ));
  context.subscriptions.push(vscode.languages.registerHoverProvider(
    [
      {language: "yaml", scheme: "file"},
      {language: "azure-pipelines", scheme: "file"}
    ],
    {
      provideHover(document, position) {
        const text = document.getText();
        if (!isAzurePipelineDocument(document.uri, text)) {
          return undefined;
        }
        const result = getExpressionHover(text, document.offsetAt(position));
        return result
          ? new vscode.Hover(new vscode.MarkdownString(`**${result.title}**\n\n${result.body}`))
          : undefined;
      }
    }
  ));

  const refreshAll = () => {
    pipelinesProvider.refresh();
    branchProvider.refresh();
    libraryProvider.refresh();
    void refreshPinnedStatus();
  };

  const statusBar = vscode.window.createStatusBarItem(vscode.StatusBarAlignment.Right, 40);
  statusBar.command = "azure-pipelines.showPinned";
  const refreshPinnedStatus = async () => {
    const pinned = getPinned();
    if (!pinned.length) {
      statusBar.hide();
      return;
    }
    const api = await getApi(false);
    const statuses = api
      ? await Promise.all(pinned.map(async pipeline => {
          try {
            const response = await api.listRuns(pipeline.id);
            const latest = responseItems(response, "runs")[0];
            return {
              name: pipeline.name,
              status: latest ? (latest.result || latest.state || "unknown") : "No runs"
            };
          } catch {
            return {name: pipeline.name, status: "Unavailable"};
          }
        }))
      : [];
    const first = statuses[0];
    statusBar.text = first
      ? `$(rocket) ${first.name}: ${first.status}`
      : `$(rocket) Azure Pipelines: ${pinned.length}`;
    statusBar.tooltip = [
      "Show pinned Azure Pipelines",
      ...statuses.map(status => `${status.name}: ${status.status}`)
    ].join("\n");
    statusBar.show();
  };
  context.subscriptions.push(statusBar);

  register("azure-pipelines.refreshSchema", async () => {
    const api = await getApi(true);
    const folder = vscode.workspace.workspaceFolders?.[0];
    if (!api || !folder) {
      if (!folder) {
        void vscode.window.showErrorMessage("Open a workspace folder before downloading an organization schema.");
      }
      return;
    }
    const schemaUri = vscode.Uri.joinPath(folder.uri, ".vscode", "azure-pipelines-schema.json");
    const answer = await vscode.window.showWarningMessage(
      "Download the organization's pipeline schema to .vscode/azure-pipelines-schema.json and use it for Azure Pipelines YAML?",
      {modal: true},
      "Download"
    );
    if (answer !== "Download") {
      return;
    }
    try {
      const schema = await api.getYamlSchema();
      await vscode.workspace.fs.createDirectory(vscode.Uri.joinPath(folder.uri, ".vscode"));
      await vscode.workspace.fs.writeFile(
        schemaUri,
        Buffer.from(JSON.stringify(schema, null, 2), "utf8")
      );
      const yamlSettings = vscode.workspace.getConfiguration("yaml");
      const mappings = {...yamlSettings.get("schemas", {})};
      delete mappings["https://raw.githubusercontent.com/microsoft/azure-pipelines-vscode/main/service-schema.json"];
      mappings[`./${vscode.workspace.asRelativePath(schemaUri)}`] = [
        "azure-pipelines.yml",
        "azure-pipelines.yaml",
        "**/azure-pipelines.yml",
        "**/azure-pipelines.yaml",
        ".azure-pipelines/**/*.yml",
        ".azure-pipelines/**/*.yaml",
        "**/.azure-pipelines/**/*.yml",
        "**/.azure-pipelines/**/*.yaml",
        "**/azure-pipelines/**/*.yml",
        "**/azure-pipelines/**/*.yaml",
        "**/.pipelines/**/*.yml",
        "**/.pipelines/**/*.yaml",
        "vsts-ci.yml",
        "vsts-ci.yaml",
        ".vsts-ci.yml",
        ".vsts-ci.yaml",
        "**/*.azure-pipelines.yml",
        "**/*.azure-pipelines.yaml"
      ];
      await yamlSettings.update("schemas", mappings, vscode.ConfigurationTarget.Workspace);
      await vscode.workspace.getConfiguration("azure-pipelines").update(
        "customSchemaFile",
        `./${vscode.workspace.asRelativePath(schemaUri)}`,
        vscode.ConfigurationTarget.Workspace
      );
      void vscode.window.showInformationMessage("Downloaded and configured the organization pipeline schema.");
    } catch (error) {
      void vscode.window.showErrorMessage(`Could not download the organization schema: ${error.message}`);
    }
  });

      return;

      pipeline = await selectPipeline(api);

  register("azure-pipelines.queuePipeline", async argument => {
    let pipeline = getPipelineArgument(argument);
    const api = await getApi(true);
    if (!api) {
      return;
    }
    if (!pipeline?.id) {
      pipeline = await selectPipeline(api);
      if (!pipeline) {
        return;
      }
    }
    const branch = await vscode.window.showInputBox({
      title: `Run ${pipeline.name}`,
      prompt: "Branch or ref to run",
      value: await currentGitBranch() || "main",
      ignoreFocusOut: true
    });
    if (!branch?.trim()) {
      return;
    }
    let templateParameters;
    let variables;
    try {
      const definition = await findPipelineYaml(api, pipeline);
      let parameterDefinitions = [];
      if (definition.uri) {
        const document = await vscode.workspace.openTextDocument(definition.uri);
        parameterDefinitions = getParameterDefinitions(document.getText());
      }
      if (parameterDefinitions.length) {
        templateParameters = await promptPipelineParameters(
          parameterDefinitions,
          "Values read from the pipeline YAML in this workspace"
        );
        if (!templateParameters) {
          return;
        }
      } else {
        const parameterText = await vscode.window.showInputBox({
          title: `Run ${pipeline.name}`,
          prompt: "Optional template parameters as a JSON object (leave blank to skip)",
          ignoreFocusOut: true
        });
        if (parameterText === undefined) {
          return;
        }
        templateParameters = parseJsonObject(parameterText, "Template parameters");
      }
      const variableText = await vscode.window.showInputBox({
        title: `Run ${pipeline.name}`,
        prompt: "Optional run variables as a JSON object (leave blank to skip)",
        password: true,
        ignoreFocusOut: true
      });
      if (variableText === undefined) {
        return;
      }
      variables = normalizeVariables(parseJsonObject(variableText, "Run variables"));
      const logging = await vscode.window.showQuickPick([
        {label: "Standard logging", value: false},
        {label: "Enable verbose diagnostics (System.Debug)", value: true}
      ], {placeHolder: "Pipeline logging"});
      if (!logging) {
        return;
      }
      if (logging.value) {
        variables = normalizeVariables({...variables, "system.debug": true});
      }
    } catch (error) {
      void vscode.window.showErrorMessage(error.message);
      return;
    }
    const answer = await vscode.window.showWarningMessage(
      `Queue pipeline "${pipeline.name}" on "${branch.trim()}"?`,
      {modal: true},
      "Queue"
    );
    if (answer !== "Queue") {
      return;
    }
    try {
      const run = await api.runPipeline(pipeline.id, {
        branch: branch.trim(),
        templateParameters,
        variables
      });
      refreshAll();
      if (run?.id) {
        void vscode.window.showInformationMessage(`Queued ${pipeline.name} run #${run.id}.`);
        await vscode.env.openExternal(vscode.Uri.parse(webRunUrl(
          (await getConnectionValues()).organization,
          (await getConnectionValues()).project,
          run
        )));
      }
    } catch (error) {
      void vscode.window.showErrorMessage(`Could not queue pipeline: ${error.message}`);
    }
  });

  register("azure-pipelines.pinPipeline", async argument => {
    const pipeline = getPipelineArgument(argument);
    if (!pipeline?.id) {
      return;

function parseJsonObject(value, title) {
  if (!value?.trim()) {
    return undefined;
  }
  let parsed;
  try {
    parsed = JSON.parse(value);
  } catch {
    throw new Error(`${title} must be valid JSON.`);
  }
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
    throw new Error(`${title} must be a JSON object.`);
  }
  return parsed;
}

function normalizeVariables(variables) {
  if (!variables) {
    return undefined;
  }
  return Object.fromEntries(Object.entries(variables).map(([name, value]) => [
    name,
    value && typeof value === "object" && !Array.isArray(value)
      ? value
      : {value: String(value)}
  ]));
}

function webRunUrl(organization, project, run) {
  return `https://dev.azure.com/${encodeURIComponent(organization)}/${encodeURIComponent(project)}/_build/results?buildId=${encodeURIComponent(run.id)}&view=results`;
}

function webLogUrl(organization, project, run, record) {
  const url = new URL(webRunUrl(organization, project, run));
  url.searchParams.set("view", "logs");
  if (record?.parentId) {
    url.searchParams.set("j", record.parentId);
  }
  if (record?.id) {
    url.searchParams.set("t", record.id);
  }
  return url.toString();
}

function statusIcon(run) {
  const state = String(run.state || "").toLowerCase();
  const result = String(run.result || "").toLowerCase();
  if (["inprogress", "cancelling", "notstarted"].includes(state)) {
    return new vscode.ThemeIcon("sync~spin", new vscode.ThemeColor("charts.blue"));
  }
  if (result === "succeeded") {
    return new vscode.ThemeIcon("pass", new vscode.ThemeColor("testing.iconPassed"));
  }
  if (result === "failed" || result === "canceled") {
    return new vscode.ThemeIcon("error", new vscode.ThemeColor("testing.iconFailed"));
  }
  if (result === "partiallysucceeded") {
    return new vscode.ThemeIcon("warning", new vscode.ThemeColor("testing.iconQueued"));
  }
  return new vscode.ThemeIcon("circle-outline");
}

function runLabel(run) {
  const name = run.name || `Run ${run.id}`;
  const status = run.state === "completed" ? (run.result || "completed") : (run.state || "unknown");
  return `#${run.id} ${name} - ${status}`;

class PipelineProvider {
  constructor({getApi, getPinnedIds, currentBranch = false, getBranch}) {
    this.getApi = getApi;
    this.getPinnedIds = getPinnedIds;
    this.currentBranch = currentBranch;
    this.getBranch = getBranch;
    this.onDidChangeTreeDataEmitter = new vscode.EventEmitter();
    this.onDidChangeTreeData = this.onDidChangeTreeDataEmitter.event;
    this.timelines = new Map();
    this.branchRuns = new Map();
  }

  refresh() {
    this.timelines.clear();
    this.branchRuns.clear();
    this.onDidChangeTreeDataEmitter.fire();
  }

  getTreeItem(element) {
    return element;
  }

  async getChildren(element) {
    const api = await this.getApi(false);
    if (!api) {
      return [informationNode("Configure Azure DevOps", "Organization, project, and authentication required", "azure-pipelines.configure")];
    }

    try {
      if (!element) {
        if (this.currentBranch) {
          return this.getPipelinesForBranch(api);
        }
        const pipelines = responseItems(await api.listPipelines(), "pipelines");
        if (!pipelines.length) {
          return [informationNode("No pipelines found", "Check the selected project and permissions")];
        }
        return pipelines.map(item => pipelineNode(item, this.getPinnedIds()));
      }

      if (element.previousAttempt && element.run && element.pipeline) {
        const timeline = await this.getTimeline(api, element.run, element.previousAttempt.timelineId);
        const record = (timeline?.records || []).find(item => item.id === element.previousAttempt.recordId);
        return record
          ? [recordNode(
              {...record, attempt: element.previousAttempt.attempt},
              (timeline.records || []).some(child => child.parentId === record.id),
              element.run,
              element.pipeline,
              element.previousAttempt.timelineId
            )]
          : [];
      }

      if (element.record && element.run && element.pipeline) {
        const timeline = await this.getTimeline(api, element.run, element.timelineId);
        const children = (timeline?.records || [])
          .filter(record => record.parentId === element.record.id)
          .map(record => recordNode(
            record,
            (timeline.records || []).some(child => child.parentId === record.id),
            element.run,
            element.pipeline,
            element.timelineId
          ));
        const previous = (element.record.previousAttempts || []).map(attempt => {
          const item = new vscode.TreeItem(
            `Previous attempt ${attempt.attempt}`,
            vscode.TreeItemCollapsibleState.Collapsed
          );
          item.id = `attempt-${element.record.id}-${attempt.attempt}`;
          item.contextValue = "azurePipelinePreviousAttempt";
          item.previousAttempt = attempt;
          item.run = element.run;
          item.pipeline = element.pipeline;
          return item;
        });
        return [...children, ...previous];
      }

      if (element.run && element.pipeline) {
        return this.getRunDetails(api, element.run, element.pipeline);
      }

      if (element.pipeline) {
        const runs = this.currentBranch
          ? (this.branchRuns.get(Number(element.pipeline.id)) || [])
          : responseItems(await api.listRuns(element.pipeline.id), "runs").slice(0, 100);
        return runs.map(run => runNode(run, element.pipeline));
      }

      if (element.log) {
        return [];
      }
    } catch (error) {
      return [informationNode("Azure DevOps request failed", error.message)];
    }
    return [];
  }

  async getPipelinesForBranch(api) {
    const branch = await this.getBranch();
    if (!branch) {
      return [informationNode("No Git branch detected", "Open a Git repository to filter runs by branch")];
    }
    const [buildsResponse, pipelinesResponse] = await Promise.all([
      api.listBuildsForBranch(branch),
      api.listPipelines()
    ]);
    const pipelineIds = new Set(
      responseItems(pipelinesResponse, "pipelines").map(pipeline => Number(pipeline.id))
    );
    const builds = responseItems(buildsResponse, "builds");
    const grouped = new Map();
    for (const build of builds) {
      if (!build.definition?.id || !pipelineIds.has(Number(build.definition.id))) {
        continue;
      }
      const pipeline = {
        id: build.definition.id,
        name: build.definition.name || `Pipeline ${build.definition.id}`
      };
      const run = {
        id: build.id,
        name: build.buildNumber || `Run ${build.id}`,
        state: build.status === "completed" ? "completed" : build.status,
        result: build.result,
        createdDate: build.startTime || build.queueTime,
        resources: {repositories: {self: {refName: build.sourceBranch}}}
      };
      const key = Number(pipeline.id);
      if (!grouped.has(key)) {
        grouped.set(key, {pipeline, runs: []});
      }
      grouped.get(key).runs.push(run);
    }
    this.branchRuns = new Map([...grouped].map(([id, item]) => [id, item.runs]));
    return [...grouped.values()].map(item => pipelineNode(item.pipeline, this.getPinnedIds()));
  }

  async getTimeline(api, run, timelineId) {
    const key = `${run.id}:${timelineId || ""}`;
    if (!this.timelines.has(key)) {
      this.timelines.set(key, api.getBuildTimeline(run.id, timelineId).catch(() => undefined));
    }
    return this.timelines.get(key);
  }

  async getRunDetails(api, run, pipeline) {
    const timeline = await this.getTimeline(api, run);
    const records = timeline?.records || [];
    if (records.length) {
      const rootRecords = records.filter(record => !record.parentId);
      return rootRecords.map(record => {
        return recordNode(record, records.some(child => child.parentId === record.id), run, pipeline);
      });
    }

    try {
      const logResponse = await api.listPipelineLogs(pipeline.id, run.id);
      const logs = responseItems(logResponse, "logs");
      return logs.map(log => {
        const item = new vscode.TreeItem(
          log.name || `Log ${log.id}`,
          vscode.TreeItemCollapsibleState.None
        );
        item.id = `log-${pipeline.id}-${run.id}-${log.id}`;
        item.contextValue = "azurePipelineLog";
        item.iconPath = new vscode.ThemeIcon("output");
        item.description = log.lineCount ? `${log.lineCount} lines` : undefined;
        item.command = {
          command: "azure-pipelines.openLogs",
          title: "Open Logs",
          arguments: [{log, run, pipeline}]
        };
        return item;
      });
    } catch {
      return [informationNode("Run details unavailable", "Timeline and logs are not available for this run")];
    }
  }
}

function pipelineNames(text) {
  const root = parsePipelineYaml(text) || {};
  const stages = [];
  const jobs = [];
  const steps = [];
  const jobsByStage = new Map();

    for (const job of Array.isArray(list) ? list : []) {
      const name = job.job || job.deployment;
      if (name) {
        jobs.push(String(name));
      }
      const jobName = String(name || "");
      const namedSteps = [];
      for (const step of job.steps || []) {
        if (step.name) {
          steps.push(String(step.name));
          namedSteps.push(String(step.name));
        }
      }
      if (jobName) {
        stepsByJob.set(jobName, namedSteps);
        if (stageName) {
          if (!jobsByStage.has(stageName)) {

function expressionAt(text, offset) {
  const before = text.slice(0, offset);
  const candidates = [
    ["${{", "}}", "template"],
    ["$[", "]", "runtime"],
    ["$(", ")", "macro"]
  ];
  let selected;
  for (const [open, close, kind] of candidates) {
    const start = before.lastIndexOf(open);
    const end = before.lastIndexOf(close);
    if (start >= 0 && start > end && (!selected || start > selected.start)) {
      selected = {start: start + open.length, kind};
    }
  }
  if (selected) {
    return {kind: selected.kind, body: before.slice(selected.start)};
  }
  const lineStart = before.lastIndexOf("\n") + 1;
  const line = before.slice(lineStart);
  if (/\bcondition\s*:\s*[^#]*$/i.test(line)) {
    return {kind: "condition", body: line.replace(/^.*?\bcondition\s*:\s*/i, "")};
  }
  return undefined;
}

function getExpressionHover(text, offset) {
  const word = text.slice(0, offset).match(/[A-Za-z_][A-Za-z0-9_.-]* *$/)?.[0]?.trim();
  if (!word) {
    return undefined;
  }
  const fn = FUNCTION_MAP.get(word);
  if (fn) {
    return {title: `Azure Pipelines expression: ${fn.signature}`, body: fn.description};
  }
  const context = CONTEXTS.find(([name]) => name === word);
  if (context) {
    return {title: `${word} context`, body: context[1]};
  }
  if (SYSTEM_VARIABLES.includes(word)) {
    return {title: `${word} variable`, body: "A predefined Azure Pipelines variable. Availability depends on the pipeline, stage, job, and expression phase."};
  }
  return undefined;
}

function getExpressionDiagnostics(text) {
  const diagnostics = [];
  const openers = [["${{", "}}"], ["$[", "]"]];
  for (const [open, close] of openers) {
    let searchFrom = 0;
    while (true) {
      const start = text.indexOf(open, searchFrom);
      if (start < 0) {
        break;
      }
      const end = text.indexOf(close, start + open.length);
      if (end < 0) {
        diagnostics.push({start, end: start + open.length, message: `Unclosed Azure Pipelines expression; expected "${close}".`});
        break;
      }
      const bodyStart = start + open.length;
      const body = text.slice(bodyStart, end);
      const calls = /\b([A-Za-z_][A-Za-z0-9_]*)\s*\(/g;
      let match;
      while ((match = calls.exec(body))) {
        if (!FUNCTION_MAP.has(match[1])) {
          const functionStart = bodyStart + match.index;
          diagnostics.push({
            start: functionStart,
            end: functionStart + match[1].length,
            message: `Unknown Azure Pipelines expression function "${match[1]}".`
          });
        }
      }
      searchFrom = end + close.length;
    }
  }
  return diagnostics;
}
