"use strict";

const API_VERSION = "7.1";

class AzureDevOpsApi {
  constructor({
    organization,
    project,
    token,
    authType = "pat",
    fetchImpl = globalThis.fetch,
    workspaceFolder
  }) {
    if (!organization || !project || !token) {
      throw new Error("Azure DevOps organization, project, and an access token are required.");
    }
    if (typeof fetchImpl !== "function") {
      throw new Error("This VS Code runtime does not provide fetch().");
    }

    this.organization = organization.trim();
    this.project = project.trim();
    this.token = token;
    this.authType = authType;
    this.workspaceFolder = workspaceFolder;
    this.fetch = fetchImpl;
    this.projectDetails = undefined;
    this.resolvedProjectId = undefined;
  }

  projectUrl(route, query = {}) {
    const base = `https://dev.azure.com/${encodeURIComponent(this.organization)}/${encodeURIComponent(this.project)}/_apis/${route}`;
    return this.withQuery(base, query);
  }

  organizationUrl(route, query = {}) {
    const base = `https://dev.azure.com/${encodeURIComponent(this.organization)}/_apis/${route}`;
    return this.withQuery(base, query);
  }

  withQuery(base, query) {
    const url = new URL(base);
    for (const [key, value] of Object.entries({...query, "api-version": API_VERSION})) {
      if (value !== undefined && value !== null && value !== "") {
        url.searchParams.set(key, String(value));
      }
    }
    return url.toString();
  }

  async request(url, {method = "GET", body, accept = "application/json"} = {}) {
    const headers = {Accept: accept};
    headers.Authorization = this.authType === "bearer"
      ? `Bearer ${this.token}`
      : `Basic ${Buffer.from(`:${this.token}`).toString("base64")}`;
    if (body !== undefined) {
      headers["Content-Type"] = "application/json";
    }

    const response = await this.fetch(url, {
      method,
      headers,
      body: body === undefined ? undefined : JSON.stringify(body)
    });
    const text = await response.text();
    if (!response.ok) {
      let detail = text;
      try {
        const parsed = JSON.parse(text);
        detail = parsed.message || parsed.typeKey || text;
      } catch {
        // Keep the raw response when Azure DevOps does not return JSON.
      }
      throw new Error(`Azure DevOps returned ${response.status}: ${detail || response.statusText}`);
    }
    if (!text) {
      return null;
    }
    try {
      return JSON.parse(text);
    } catch {
      return text;
    }
  }

  listPipelines() {
    return this.request(this.projectUrl("pipelines", {$top: 200}));
  }

  listBuildDefinitions() {
    return this.request(this.projectUrl("build/definitions", {$top: 200}));
  }

  getBuildDefinition(definitionId) {
    return this.request(this.projectUrl(
      `build/definitions/${encodeURIComponent(definitionId)}`
    ));
  }

  updateBuildDefinition(definition) {
    return this.request(this.projectUrl(
      `build/definitions/${encodeURIComponent(definition.id)}`
    ), {
      method: "PUT",
      body: definition
    });
  }

  async getProject() {
    if (!this.projectDetails) {
      this.projectDetails = await this.request(
        this.organizationUrl(`projects/${encodeURIComponent(this.project)}`)
      );
      this.resolvedProjectId = this.projectDetails.id;
    }
    return this.projectDetails;
  }

  async getProjectId() {
    await this.getProject();
    return this.resolvedProjectId;
  }

  getPipeline(pipelineId) {
    return this.request(this.projectUrl(`pipelines/${encodeURIComponent(pipelineId)}`));
  }

  listRuns(pipelineId) {
    return this.request(this.projectUrl(`pipelines/${encodeURIComponent(pipelineId)}/runs`));
  }

  listBuildsForBranch(branch) {
    const branchName = branch.startsWith("refs/") ? branch : `refs/heads/${branch}`;
    return this.request(this.projectUrl("build/builds", {
      branchName,
      $top: 100,
      queryOrder: "queueTimeDescending"
    }));
  }

  runPipeline(pipelineId, {branch, templateParameters, variables} = {}) {
    const body = {
      resources: branch
        ? {repositories: {self: {refName: branch.startsWith("refs/") ? branch : `refs/heads/${branch}`}}}
        : undefined,
      templateParameters: templateParameters || undefined,
      variables: variables || undefined
    };
    return this.request(this.projectUrl(`pipelines/${encodeURIComponent(pipelineId)}/runs`), {
      method: "POST",
      body
    });
  }

  listPipelineLogs(pipelineId, runId) {
    return this.request(this.projectUrl(
      `pipelines/${encodeURIComponent(pipelineId)}/runs/${encodeURIComponent(runId)}/logs`
    ));
  }

  async getPipelineLog(pipelineId, runId, logId) {
    const response = await this.request(this.projectUrl(
      `pipelines/${encodeURIComponent(pipelineId)}/runs/${encodeURIComponent(runId)}/logs/${encodeURIComponent(logId)}`,
      {$expand: "signedContent"}
    ), {accept: "text/plain"});
    if (typeof response !== "object" || response === null) {
      return response;
    }
    const signedContent = response.signedContent;
    const signedUrl = typeof signedContent === "string"
      ? signedContent
      : signedContent?.url || response.url;
    if (signedUrl) {
      const url = new URL(signedUrl);
      if (url.protocol !== "https:") {
        throw new Error("Azure DevOps returned a non-HTTPS log URL.");
      }
      const result = await this.fetch(url, {headers: {Accept: "text/plain"}});
      if (!result.ok) {
        throw new Error(`Could not download the log content (${result.status}).`);
      }
      return result.text();
    }
    if (typeof signedContent?.content === "string") {
      return signedContent.content;
    }
    return response;
  }

  getYamlSchema() {
    return this.request(this.organizationUrl("distributedtask/yamlschema"));
  }

  listTaskDefinitions() {
    return this.request(this.organizationUrl("distributedtask/tasks"));
  }

  getBuildTimeline(runId, timelineId) {
    const timeline = timelineId ? `/timeline/${encodeURIComponent(timelineId)}` : "/timeline";
    return this.request(this.projectUrl(`build/builds/${encodeURIComponent(runId)}${timeline}`));
  }

  getBuild(runId) {
    return this.request(this.projectUrl(`build/builds/${encodeURIComponent(runId)}`));
  }

  getBuildLog(runId, logId, startLine) {
    return this.request(this.projectUrl(
      `build/builds/${encodeURIComponent(runId)}/logs/${encodeURIComponent(logId)}`,
      startLine === undefined ? {} : {startLine}
    ), {accept: "text/plain"});
  }

  listBuildArtifacts(runId) {
    return this.request(this.projectUrl(
      `build/builds/${encodeURIComponent(runId)}/artifacts`
    ));
  }

  async downloadBuildArtifact(runId, artifactName) {
    const url = this.projectUrl(
      `build/builds/${encodeURIComponent(runId)}/artifacts`,
      {artifactName, "$format": "zip"}
    );
    const authorization = this.authType === "bearer"
      ? `Bearer ${this.token}`
      : `Basic ${Buffer.from(`:${this.token}`).toString("base64")}`;
    const response = await this.fetch(url, {
      headers: {Authorization: authorization, Accept: "application/zip"}
    });
    if (!response.ok) {
      throw new Error(`Could not download the artifact (${response.status}).`);
    }
    return Buffer.from(await response.arrayBuffer());
  }

  retryStage(runId, stageRefName, {forceRetryAllJobs = false} = {}) {
    return this.request(this.projectUrl(
      `build/builds/${encodeURIComponent(runId)}/stages/${encodeURIComponent(stageRefName)}`
    ), {
      method: "PATCH",
      body: {state: "retry", forceRetryAllJobs}
    });
  }

  listPendingApprovals() {
    return this.request(this.projectUrl("pipelines/approvals", {
      state: "pending",
      "$expand": "steps"
    }));
  }

  updateApproval(approvalId, status, comment) {
    return this.request(this.projectUrl("pipelines/approvals"), {
      method: "PATCH",
      body: [{
        approvalId,
        status,
        comment: comment || undefined
      }]
    });
  }

  cancelRun(runId) {
    return this.request(this.projectUrl(`build/builds/${encodeURIComponent(runId)}`), {
      method: "PATCH",
      body: {id: Number(runId), status: "cancelling"}
    });
  }

  retryRun(runId) {
    return this.request(this.projectUrl(`build/builds/${encodeURIComponent(runId)}`, {retry: true}), {
      method: "PATCH",
      body: {id: Number(runId)}
    });
  }

  listEnvironments() {
    return this.request(this.projectUrl("distributedtask/environments"));
  }

  createEnvironment(environment) {
    return this.request(this.projectUrl("distributedtask/environments"), {
      method: "POST",
      body: environment
    });
  }

  updateEnvironment(environmentId, environment) {
    return this.request(this.projectUrl(
      `distributedtask/environments/${encodeURIComponent(environmentId)}`
    ), {
      method: "PATCH",
      body: environment
    });
  }

  deleteEnvironment(environmentId) {
    return this.request(this.projectUrl(
      `distributedtask/environments/${encodeURIComponent(environmentId)}`
    ), {method: "DELETE"});
  }

  async createVariableGroup(name, description = "") {
    const project = await this.getProject();
    return this.request(this.organizationUrl("distributedtask/variablegroups"), {
      method: "POST",
      body: {
        name,
        description,
        type: "Vsts",
        variables: {},
        variableGroupProjectReferences: [{
          name,
          description,
          projectReference: {id: project.id, name: project.name}
        }]
      }
    });
  }

  async deleteVariableGroup(groupId) {
    const projectId = await this.getProjectId();
    return this.request(this.organizationUrl(
      `distributedtask/variablegroups/${encodeURIComponent(groupId)}`,
      {projectIds: projectId}
    ), {method: "DELETE"});
  }

  async listVariableGroups() {
    const projectId = await this.getProjectId();
    return this.request(this.organizationUrl("distributedtask/variablegroups", {
      projectIds: projectId
    }));
  }

  async getVariableGroup(groupId) {
    const projectId = await this.getProjectId();
    return this.request(this.organizationUrl(
      `distributedtask/variablegroups/${encodeURIComponent(groupId)}`,
      {projectIds: projectId}
    ));
  }

  async updateVariableGroup(group) {
    const projectId = await this.getProjectId();
    const {id, ...body} = group;
    return this.request(this.organizationUrl(
      `distributedtask/variablegroups/${encodeURIComponent(id)}`,
      {projectIds: projectId}
    ), {
      method: "PUT",
      body
    });

async function currentGitBranch(folder) {
  const folders = vscode.workspace.workspaceFolders || [];
  const target = folder || activeWorkspaceFolder() || (folders.length === 1 ? folders[0] : undefined);
  if (!target) {
    return undefined;
  }

  let repositoryRoot;
  try {
    const gitExtension = vscode.extensions.getExtension("vscode.git");
    if (gitExtension) {
      const git = gitExtension.isActive
        ? gitExtension.exports
        : await gitExtension.activate();
      const api = git.getAPI(1);
      const root = target.uri.fsPath.toLowerCase();
      const repositories = api.repositories.map(item => ({
        item,
        root: item.rootUri.fsPath.toLowerCase()
      }));
      const ancestors = repositories.filter(repository =>
        root === repository.root || root.startsWith(`${repository.root}${path.sep.toLowerCase()}`)
      ).sort((left, right) => right.root.length - left.root.length);
      const descendants = repositories.filter(repository =>
        repository.root.startsWith(`${root}${path.sep.toLowerCase()}`)
      );
      const repository = ancestors[0]?.item || (descendants.length === 1 ? descendants[0].item : undefined);
      if (repository?.state.HEAD?.name) {
        return repository.state.HEAD.name;
      }
      repositoryRoot = repository?.rootUri.fsPath;
    }
  } catch {
    // Fall back to Git CLI when the built-in Git extension is unavailable.
  }

  try {
    const {stdout} = await execFileAsync("git", ["branch", "--show-current"], {
      cwd: repositoryRoot || target.uri.fsPath,
      windowsHide: true,
      timeout: 3000
    });
    return stdout.trim() || undefined;
  } catch {
    return undefined;
  }
}

  register("azure-pipelines.queuePipeline", async argument => {
    let pipeline = getPipelineArgument(argument);
    const api = await getApi(true, pipeline?.workspaceFolder);
    if (!api) {
      return;
    }
    if (!pipeline?.id) {
      pipeline = await selectPipeline(api);
      if (!pipeline) {
        return;
      }
    }
    const branch = await vscode.window.showInputBox({
      title: `Run ${pipeline.name}`,
      prompt: "Branch or ref to run",
      value: await currentGitBranch(pipeline.workspaceFolder) || "main",
      ignoreFocusOut: true
    });
    if (!branch?.trim()) {
      return;
    }
    let templateParameters;
    let variables;
    try {
      const definition = await findPipelineYaml(api, pipeline);
      let parameterDefinitions = [];
      if (definition.uri) {
        const document = await vscode.workspace.openTextDocument(definition.uri);
        parameterDefinitions = getParameterDefinitions(document.getText());
      }
      if (parameterDefinitions.length) {
        templateParameters = await promptPipelineParameters(
          parameterDefinitions,
          "Values read from the pipeline YAML in this workspace"
        );
        if (!templateParameters) {
          return;
        }
      } else {
        const parameterText = await vscode.window.showInputBox({
          title: `Run ${pipeline.name}`,
          prompt: "Optional template parameters as a JSON object (leave blank to skip)",
          ignoreFocusOut: true
        });
        if (parameterText === undefined) {
          return;
        }
        templateParameters = parseJsonObject(parameterText, "Template parameters");
      }
      const variableText = await vscode.window.showInputBox({
        title: `Run ${pipeline.name}`,
        prompt: "Optional run variables as a JSON object (leave blank to skip)",
        ignoreFocusOut: true
      });
      if (variableText === undefined) {
        return;
      }
      variables = normalizeVariables(parseJsonObject(variableText, "Run variables"));
      const logging = await vscode.window.showQuickPick([
        {label: "Standard logging", value: false},
        {label: "Enable verbose diagnostics (System.Debug)", value: true}
      ], {placeHolder: "Pipeline logging"});
      if (!logging) {
        return;
      }
      if (logging.value) {
        variables = normalizeVariables({...variables, "system.debug": true});
      }
    } catch (error) {
      void vscode.window.showErrorMessage(error.message);
      return;
    }
    const answer = await vscode.window.showWarningMessage(
      `Queue pipeline "${pipeline.name}" on "${branch.trim()}"?`,
      {modal: true},
      "Queue"
    );
    if (answer !== "Queue") {
      return;
    }
    try {
      const run = await api.runPipeline(pipeline.id, {
        branch: branch.trim(),
        templateParameters,
        variables
      });
      refreshAll();
      if (run?.id) {
        void vscode.window.showInformationMessage(`Queued ${pipeline.name} run #${run.id}.`);
        const values = await getConnectionValues(pipeline.workspaceFolder);
        await vscode.env.openExternal(vscode.Uri.parse(webRunUrl(
          values.organization,
          values.project,
          run
        )));
      }
    } catch (error) {
      void vscode.window.showErrorMessage(`Could not queue pipeline: ${error.message}`);
    }
  });

  register("azure-pipelines.openLogs", async argument => {
    const run = argument?.run || argument?.record?.run;
    const pipeline = argument?.pipeline || argument?.record?.pipeline;
    const log = argument?.log;
    const record = argument?.record;
    const logId = record?.log?.id ?? log?.id;
    const api = await getApi(true, pipeline?.workspaceFolder);
    if (!api || logId === undefined || !run) {
      return;
    }
    try {
      const fetchContent = async startLine => {
        let content = record
          ? await api.getBuildLog(run.id, logId, startLine)
          : await api.getPipelineLog(pipeline.id, run.id, logId);
        if (content && typeof content === "object") {
          content = content.signedContent || content.url || JSON.stringify(content, null, 2);
        }
        return String(content || "");
      };
      let content = await fetchContent();
      const panel = vscode.window.createWebviewPanel(
        "azurePipelinesLog",
        `Run #${run.id} - ${record?.name || log?.name || "Log"}`,
        vscode.ViewColumn.Active,
        {enableScripts: true, retainContextWhenHidden: true}
      );
      const nonce = crypto.randomBytes(16).toString("base64");
      panel.webview.html = renderLogHtml(content, {
        cspSource: panel.webview.cspSource,
        nonce
      });

      const jumpToError = async message => {
        const file = message.file;
        const folder = pipeline?.workspaceFolder;
        if (!file) {
          return;
        }
        const roots = folder ? [folder] : (vscode.workspace.workspaceFolders || []);
        const normalizedFile = file.replace(/\\/g, "/");
        const checkoutIndex = normalizedFile.lastIndexOf("/s/");
        const candidates = [
          ...(path.isAbsolute(file) ? [path.resolve(file)] : []),
          ...(checkoutIndex >= 0 ? [normalizedFile.slice(checkoutIndex + 3)] : []),
          ...(!path.isAbsolute(file) ? [file] : [])
        ];
        const targets = [...new Set(candidates.flatMap(candidate =>
          path.isAbsolute(candidate)
            ? [path.resolve(candidate)]
            : roots.map(root => path.resolve(root.uri.fsPath, candidate))
        ))];
        for (const target of targets) {
          const root = roots.find(candidate => {
            const relative = path.relative(candidate.uri.fsPath, target);
            return relative === "" || (!relative.startsWith(`..${path.sep}`) &&
              relative !== ".." && !path.isAbsolute(relative));
          });
          if (!root) {
            continue;
          }
          try {
            const uri = vscode.Uri.file(target);
            const stat = await vscode.workspace.fs.stat(uri);
            if (!(stat.type & vscode.FileType.File)) {
              continue;
            }
            const document = await vscode.workspace.openTextDocument(uri);
            const line = Math.min(
              Math.max(0, (message.line || 1) - 1),
              Math.max(0, document.lineCount - 1)
            );
            const column = Math.min(
              Math.max(0, (message.column || 1) - 1),
              document.lineAt(line).text.length
            );
            const position = new vscode.Position(line, column);
            await vscode.window.showTextDocument(document, {
              selection: new vscode.Range(position, position)
            });
            return;
          } catch {
            // Continue through workspace roots until a matching local file is found.
          }
        }
        void vscode.window.showInformationMessage(`Source file from the build log was not found in this workspace: ${file}`);
      };
      const messageSubscription = panel.webview.onDidReceiveMessage(message => {
        if (message.type === "jump") {
          void jumpToError(message);
        }
      });
      panel.onDidDispose(() => messageSubscription.dispose());

      let polling = false;
      let timer;
      const update = async () => {
        if (polling) {
          return;
        }
        polling = true;
        try {
          const lineCount = content.split(/\r?\n/).length - (/\r?\n$/.test(content) ? 1 : 0);
          const [build, latest] = await Promise.all([
            api.getBuild(run.id).catch(() => undefined),
            fetchContent(record ? Math.max(0, lineCount - 1) : undefined)
          ]);
          if (record) {
            const merged = mergeLogContent(content, latest);
            if (merged !== undefined) {
              content = merged;
            } else {
              const complete = await fetchContent();
              if (complete.length >= content.length) {
                content = complete;
              }
            }
          } else if (latest.length > content.length && latest.startsWith(content)) {
            content = latest;
          } else if (latest !== content && latest.length >= content.length) {
            content = latest;
          }
          const active = build ? build.status !== "completed" : run.state !== "completed";
          await panel.webview.postMessage({
            type: "update",
            logHtml: renderLogBody(content),
            errorsHtml: renderErrors(content),
            active
          });
          if (!active && timer) {
            clearInterval(timer);
            timer = undefined;
          }
        } catch {
          // A temporary network failure does not close the log viewer.
        } finally {
          polling = false;
        }
      };
      if (run.state !== "completed") {
        timer = setInterval(() => void update(), 5000);
        panel.onDidDispose(() => clearInterval(timer));
      }
    } catch (error) {
      void vscode.window.showErrorMessage(`Could not load log: ${error.message}`);
    }
  });

  const mutateVariableGroup = async (groupNode, mutate) => {
    const api = await getApi(true, groupNode?.workspaceFolder);
    if (!api || !groupNode?.group) {
      return;
    }
    try {
      const group = await api.getVariableGroup(groupNode.group.id);
      if (group.type && group.type !== "Vsts") {
        void vscode.window.showInformationMessage("Only standard Azure DevOps variable groups can be edited here.");
        return;
      }
      const variables = {...(group.variables || {})};
      const changed = await mutate(variables, group);
      if (!changed) {
        return;
      }
      if (Object.values(group.variables || {}).some(variable => variable.isSecret && variable.value == null)) {
        const answer = await vscode.window.showWarningMessage(
          "Azure DevOps masks existing secret values. Continue updating this group? Unchanged secret values will be sent without their hidden value.",
          {modal: true},
          "Continue"
        );
        if (answer !== "Continue") {
          return;
        }
      }
      const safeVariables = Object.fromEntries(Object.entries(variables).map(([name, variable]) => {
        if (variable.isSecret && variable.value == null) {
          const {value, ...maskedSecret} = variable;
          return [name, maskedSecret];
        }

function parseAzureDevOpsRemote(remote) {
  if (typeof remote !== "string" || !remote.trim()) {
    return undefined;
  }

  const value = remote.trim();
  let host;
  let path;
  try {
    const normalized = /^[^/]+@[^:]+:/.test(value)
      ? `ssh://${value.replace(":", "/")}`
      : value;
    const url = new URL(normalized);
    host = url.hostname.toLowerCase();
    path = decodeURIComponent(url.pathname).split("/").filter(Boolean);
  } catch {
    return undefined;
  }

  let organization;
  let project;
  let repository;
  if (host === "dev.azure.com") {
    [organization, project] = path;
    if (path[2]?.toLowerCase() === "_git") {
      repository = path[3];
    }
  } else if (
    host === "ssh.dev.azure.com" ||
    host === "vs-ssh.visualstudio.com" ||
    host.endsWith(".vs-ssh.visualstudio.com")
  ) {
    const v3Index = path.findIndex(segment => segment.toLowerCase() === "v3");
    if (v3Index >= 0) {
      [organization, project, repository] = path.slice(v3Index + 1);
    }
  } else if (host.endsWith(".visualstudio.com")) {
    organization = host.slice(0, -".visualstudio.com".length);
    const collectionIndex = path[0]?.toLowerCase() === "defaultcollection" ? 1 : 0;
    project = path[collectionIndex];
    if (path[collectionIndex + 1]?.toLowerCase() === "_git") {
      repository = path[collectionIndex + 2];
    }
  }

  if (!organization || !project) {
    return undefined;
  }
  return {organization, project, repository};
}

function parseGitRemotes(output) {
  const rows = [];
  for (const line of String(output || "").split(/\r?\n/)) {
    const configEntry = line.match(/^remote\.(.+)\.url\s+(.+)$/i);
    const listingEntry = line.match(/^(\S+)\s+(.+?)\s+\((?:fetch|push)\)$/i);
    const name = configEntry?.[1] || listingEntry?.[1];
    const url = configEntry?.[2] || listingEntry?.[2] || line;
    const remote = parseAzureDevOpsRemote(url);
    if (remote) {
      rows.push({name, remote});
    }
  }

  const originRows = rows.filter(row => row.name === "origin");
  const candidates = originRows.length ? originRows : rows;
  const unique = new Map();
  for (const {remote: candidate} of candidates) {
    const key = `${candidate.organization.toLowerCase()}/${candidate.project.toLowerCase()}`;
    if (!unique.has(key)) {
      unique.set(key, candidate);
    }
  }
  return [...unique.values()];
}

"use strict";

function escapeHtml(value) {
  return String(value).replace(/[&<>"']/g, character => ({
    "&": "&amp;",
    "<": "&lt;",
    ">": "&gt;",
    "\"": "&quot;",
    "'": "&#39;"
  })[character]);
}

function ansiToHtml(value) {
  const colors = new Map([
    [30, "black"], [31, "red"], [32, "green"], [33, "yellow"],
    [34, "blue"], [35, "magenta"], [36, "cyan"], [37, "white"],
    [90, "bright-black"], [91, "bright-red"], [92, "bright-green"],
    [93, "bright-yellow"], [94, "bright-blue"], [95, "bright-magenta"],
    [96, "bright-cyan"], [97, "bright-white"]
  ]);
  let classes = [];
  let html = "";
  const parts = String(value).split(/\u001b\[([0-9;]*)m/);
  for (let index = 0; index < parts.length; index++) {
    if (index % 2 === 0) {
      const text = escapeHtml(parts[index]);
      html += classes.length ? `<span class="${classes.join(" ")}">${text}</span>` : text;
      continue;
    }
    const codes = parts[index] ? parts[index].split(";").map(Number) : [0];
    for (const code of codes) {
      if (code === 0) {
        classes = [];
      } else if (code === 1) {
        classes = classes.filter(name => name !== "bold");
        classes.push("bold");
      } else if (colors.has(code)) {
        classes = classes.filter(name => !name.startsWith("fg-"));
        classes.push(`fg-${colors.get(code)}`);
      } else if (colors.has(code - 10)) {
        classes = classes.filter(name => !name.startsWith("bg-"));
        classes.push(`bg-${colors.get(code - 10)}`);
      }
    }
  }
  return html;
}

function parseLogErrors(content) {
  const errors = [];
  const lines = String(content).split(/\r?\n/);
  for (let index = 0; index < lines.length; index++) {
    const line = lines[index];
    const issue = line.match(/##vso\[task\.logissue\s+([^\]]*)\](.*)/i);
    if (issue && /(?:^|;)type=error(?:;|$)/i.test(issue[1])) {
      const properties = Object.fromEntries(issue[1].split(";").map(entry => {
        const separator = entry.indexOf("=");
        return separator < 0
          ? [entry.trim().toLowerCase(), ""]
          : [entry.slice(0, separator).trim().toLowerCase(), entry.slice(separator + 1).trim()];
      }));
      errors.push({
        message: issue[2] || "Pipeline error",
        file: properties.sourcepath,
        line: Number(properties.linenumber) || undefined,
        column: Number(properties.columnnumber) || undefined,
        logLine: index + 1
      });
      continue;
    }
    const errorText = line.replace(/^##\[error\]\s*/i, "");
    const compiler = errorText.match(/^(.+?)\((\d+)(?:,(\d+))?\):\s*error\b(.*)$/i);
    if (compiler) {
      errors.push({
        message: `error${compiler[4]}`,
        file: compiler[1],
        line: Number(compiler[2]),
        column: Number(compiler[3]) || undefined,
        logLine: index + 1
      });
      continue;
    }
    if (/##\[error\]/i.test(line)) {
      errors.push({
        message: line.replace(/^.*?##\[error\]\s*/i, "") || "Pipeline error",
        logLine: index + 1
      });
    }
  }
  return errors;
}

function mergeLogContent(current, tail) {
  const existing = String(current || "");
  const update = String(tail || "");
  if (!update) {
    return existing;
  }
  if (!existing || update.startsWith(existing)) {
    return update;
  }
  if (existing.startsWith(update)) {
    return existing;
  }

  const currentLines = existing.split(/\r?\n/);
  const tailLines = update.split(/\r?\n/);
  if (currentLines[currentLines.length - 1] === "") {
    currentLines.pop();
  }
  if (tailLines[tailLines.length - 1] === "") {
    tailLines.pop();
  }
  const maxOverlap = Math.min(currentLines.length, tailLines.length);
  let overlap = 0;
  for (let size = maxOverlap; size > 0; size--) {
    if (currentLines.slice(-size).every((line, index) => line === tailLines[index])) {
      overlap = size;
      break;
    }
  }
  if (!overlap) {
    return undefined;
  }

  const appended = tailLines.slice(overlap).join("\n");
  if (!appended) {
    return existing;
  }
  let merged = `${existing}${existing.endsWith("\n") ? "" : "\n"}${appended}`;
  if (update.endsWith("\n") && !merged.endsWith("\n")) {
    merged += "\n";
  }
  return merged;
}

function renderLogBody(content) {
  const groups = [];
  const output = [];
  const lines = String(content).split(/\r?\n/);
  lines.forEach((line, index) => {
    const groupStart = line.match(/##\[group\](.*)/i);
    if (groupStart) {
      groups.push("</details>");
      output.push(`<details class="log-group" open><summary>${ansiToHtml(groupStart[1])}</summary>`);
      return;
    }
    if (/##\[endgroup\]/i.test(line)) {
      if (groups.length) {
        output.push(groups.pop());
      }
      return;
    }
    if (!line) {
      output.push(`<div class="log-line" id="log-line-${index + 1}">&nbsp;</div>`);
      return;
    }
    const error = /##\[error\]|type=error/i.test(line);
    const section = line.match(/##\[section\](.*)/i);
    const text = section ? section[1] : line;
    output.push(
      `<div class="log-line${error ? " error" : ""}" id="log-line-${index + 1}">` +
      `<span class="line-number">${index + 1}</span>${ansiToHtml(text)}</div>`
    );
  });
  while (groups.length) {
    output.push(groups.pop());
  }
  return output.join("\n");
}

function responseItems(response, key) {
  if (Array.isArray(response)) {
    return response;
  }
  return response?.[key] || response?.value || [];
}

function orderedRecords(records) {
  return records
    .map((record, index) => ({record, index, order: Number(record.order)}))
    .sort((left, right) => {
      const leftOrder = Number.isFinite(left.order) ? left.order : Number.MAX_SAFE_INTEGER;
      const rightOrder = Number.isFinite(right.order) ? right.order : Number.MAX_SAFE_INTEGER;
      return leftOrder - rightOrder || left.index - right.index;
    })
    .map(item => item.record);
}

function statusIcon(run) {
  const state = String(run.state || "").toLowerCase();
  const result = String(run.result || "").toLowerCase();
  if ([
    "inprogress",
    "cancelling",
    "canceling",
    "notstarted",
    "queued",
    "pending",
    "postponed"
  ].includes(state)) {
    return new vscode.ThemeIcon("sync~spin", new vscode.ThemeColor("charts.blue"));
  }
  if (result === "succeeded") {
    return new vscode.ThemeIcon("pass", new vscode.ThemeColor("testing.iconPassed"));
  }
  if (result === "failed" || result === "canceled") {
    return new vscode.ThemeIcon("error", new vscode.ThemeColor("testing.iconFailed"));
  }
  if (result === "partiallysucceeded") {
    return new vscode.ThemeIcon("warning", new vscode.ThemeColor("testing.iconQueued"));
  }
  return new vscode.ThemeIcon("circle-outline");
}

function findExpressionEnd(text, bodyStart, close) {
  let quote;
  let nested = 0;
  for (let index = bodyStart; index < text.length; index += 1) {
    const character = text[index];
    if (quote) {
      if (character === quote) {
        if (text[index + 1] === quote) {
          index += 1;
        } else {
          quote = undefined;
        }
      } else if (character === "\\" && index + 1 < text.length) {
        index += 1;
      }
      continue;
    }
    if (character === "'" || character === "\"") {
      quote = character;
      continue;
    }
    if (close === "]" && character === "[") {
      nested += 1;
      continue;
    }
    if (close === ")" && character === "(") {
      nested += 1;
      continue;
    }
    if ((close === "]" && character === "]") || (close === ")" && character === ")")) {
      if (nested > 0) {
        nested -= 1;
      } else {
        return index;
      }
      continue;
    }
    if (close !== "]" && close !== ")" && text.startsWith(close, index)) {
      return index;
    }
  }
  return -1;
}

function expressionRanges(text) {

  return ranges;
}

  }
  const dependencyProperties = prefix.match(/\bdependencies\.([A-Za-z0-9_-]+)\.$/);
  if (dependencyProperties) {
    return ["result", "outputs"].map(label => ({
      label,
      insertText: label,
      detail: "Dependency property"
    }));
  }
  const stageJobs = prefix.match(/\bstageDependencies\.([A-Za-z0-9_-]+)\.$/);
  if (stageJobs) {
    return (jobsByStage.get(stageJobs[1]) || []).map(label => ({
      label,
      insertText: label,
      detail: "Job in dependent stage"
    }));
  }
  const context = prefix.match(/\b(stageDependencies|dependencies|parameters|variables|resources)\.([A-Za-z0-9_.-]*)$/);
  if (context) {
    const kind = context[1];
    const names = kind === "variables" ? [...new Set([...variables, ...SYSTEM_VARIABLES])]
      : kind === "parameters" ? parameters
        : kind === "dependencies" ? [...new Set([...stages, ...jobs])]
          : kind === "stageDependencies" ? stages
            : ["pipelines", "repositories", "containers", "packages"];
    return names.map(label => ({label, insertText: label, detail: `${kind} context`}));
  }
  if (/\bsteps\.[A-Za-z0-9_-]*$/.test(prefix)) {
    return steps.map(label => ({label, insertText: label, detail: "Named pipeline step"}));
  }
  const candidates = [
    ...EXPRESSION_FUNCTIONS.map(fn => ({
      label: `${fn.name}()`,
      insertText: `${fn.name}()`,
      detail: fn.signature,
      documentation: fn.description
    })),
    ...CONTEXTS.map(([label, detail]) => ({label: `${label}.`, insertText: `${label}.`, detail})),
    {label: "true", insertText: "true", detail: "Boolean literal"},
    {label: "false", insertText: "false", detail: "Boolean literal"},
    {label: "null", insertText: "null", detail: "Null literal"}
  ];
  return candidates;
}

function getExpressionHover(text, offset) {
  const lineStart = text.lastIndexOf("\n", offset - 1) + 1;
  const lineEnd = text.indexOf("\n", offset);
  const line = text.slice(lineStart, lineEnd < 0 ? text.length : lineEnd);
  const condition = line.match(/^\s*(?:-\s*)?condition\s*:\s*/i);
  const ranges = expressionRanges(text);
  if (condition) {
    const start = lineStart + condition[0].length;
    ranges.push({
      start,
      end: lineEnd < 0 ? text.length : lineEnd
    });
  }
  const range = ranges.find(candidate => offset >= candidate.start && offset <= candidate.end);
  if (!range) {
    return undefined;
  }
  const body = text.slice(range.start, range.end);
  const maskedBody = maskStringLiterals(body);
  const localOffset = offset - range.start;
  const tokens = /[A-Za-z_][A-Za-z0-9_.-]*/g;
  let match;
  let word;
  let tokenEnd;
  while ((match = tokens.exec(maskedBody))) {
    const end = match.index + match[0].length;
    if (localOffset >= match.index && localOffset <= end) {
      word = match[0];
      tokenEnd = end;
      break;
    }
  }
  if (!word) {
    const literalTokens = /[A-Za-z_][A-Za-z0-9_.-]*/g;
    while ((match = literalTokens.exec(body))) {
      const end = match.index + match[0].length;
      if (
        localOffset >= match.index &&
        localOffset <= end &&
        SYSTEM_VARIABLES.includes(match[0])
      ) {
        return {
          title: `${match[0]} variable`,
          body: "A predefined Azure Pipelines variable. Availability depends on the pipeline, stage, job, and expression phase."
        };
      }
    }
    return undefined;
  }
  const fn = FUNCTION_MAP.get(word);
  if (fn && /^\s*\(/.test(body.slice(tokenEnd))) {
    return {title: `Azure Pipelines expression: ${fn.signature}`, body: fn.description};
  }
  const context = CONTEXTS.find(([name]) => name === word);
  if (context) {
    return {title: `${word} context`, body: context[1]};
  }
  if (SYSTEM_VARIABLES.includes(word)) {
    return {title: `${word} variable`, body: "A predefined Azure Pipelines variable. Availability depends on the pipeline, stage, job, and expression phase."};
  }
  return undefined;
}

function tokenizeExpression(body, baseOffset, diagnostics) {
  const tokens = [];
  let index = 0;
  while (index < body.length) {
    const character = body[index];
    if (/\s/.test(character)) {
      index += 1;
      continue;
    }
    if (character === "'" || character === "\"") {
      const start = index;
      const quote = character;
      index += 1;
      let closed = false;
      while (index < body.length) {
        if (body[index] === quote) {
          if (quote === "'" && body[index + 1] === quote) {
            index += 2;
          } else {
            index += 1;
            closed = true;
            break;
          }
        } else if (quote === "\"" && body[index] === "\\" && index + 1 < body.length) {
          index += 2;
        } else {
          index += 1;
        }
      }
      if (!closed) {
        diagnostics.push({
          start: baseOffset + start,
          end: baseOffset + body.length,
          message: "Unclosed string literal in Azure Pipelines expression."
        });
      }
      tokens.push({type: "string", value: body.slice(start, index), start, end: index});
      continue;
    }
    if (/[A-Za-z_]/.test(character)) {
      const start = index;
      index += 1;
      while (index < body.length && /[A-Za-z0-9_-]/.test(body[index])) {
        index += 1;
      }
      tokens.push({type: "identifier", value: body.slice(start, index), start, end: index});
      continue;
    }
    if (/\d/.test(character) || ((character === "-" || character === ".") && /\d/.test(body[index + 1] || ""))) {
      const start = index;
      const number = body.slice(index).match(/^-?(?:\d+(?:\.\d*)?|\.\d+)/)[0];
      index += number.length;
      tokens.push({type: "number", value: number, start, end: index});
      continue;
    }
    if ("()[],.".includes(character)) {
      tokens.push({type: character, value: character, start: index, end: index + 1});
      index += 1;
      continue;
    }
    tokens.push({type: "invalid", value: character, start: index, end: index + 1});
    index += 1;
  }
  return tokens;
}

function parseExpressionBody(body, baseOffset) {
  const diagnostics = [];
  const tokens = tokenizeExpression(body, baseOffset, diagnostics);
  let index = 0;
  const peek = () => tokens[index];
  const report = (message, token = peek()) => {
    const start = token ? token.start : body.length;
    const end = token ? token.end : body.length;
    diagnostics.push({start: baseOffset + start, end: baseOffset + end, message});
  };
  const reportUnknownFunction = token => {
    const known = EXPRESSION_FUNCTIONS.some(
      fn => fn.name.toLowerCase() === token.value.toLowerCase()
    );
    if (!known) {
      diagnostics.push({
        start: baseOffset + token.start,
        end: baseOffset + token.end,
        message: `Unknown Azure Pipelines expression function "${token.value}".`
      });
    }
  };
  const synchronize = closing => {
    while (peek() && peek().type !== "," && peek().type !== closing) {
      index += 1;
    }
  };
  let parseExpression;
  const parsePostfix = () => {
    while (peek()) {
      if (peek().type === ".") {
        index += 1;
        if (peek()?.type === "identifier") {
          index += 1;
        } else {
          report("Expected a property name after '.'.");
          return;
        }
      } else if (peek().type === "[") {
        index += 1;
        if (peek()?.type === "]") {
          report("Expected an index expression.");
          index += 1;
          continue;
        }
        parseExpression();
        if (peek()?.type === "]") {
          index += 1;
        } else {
          report("Expected ']' to close the index expression.");
          return;
        }
      } else {
        return;
      }
    }
  };
  const parseCallArguments = () => {
    if (peek()?.type === ")") {
      index += 1;
      return;
    }
    while (peek()) {
      if (peek().type === "," || peek().type === ")") {
        report("Expected an expression argument.");
        if (peek().type === ")") {
          index += 1;
          return;
        }
        index += 1;
        continue;
      }
      parseExpression();
      if (peek()?.type === ",") {
        index += 1;
        if (peek()?.type === ")") {
          report("Expected an expression argument.");
          index += 1;
          return;
        }
        continue;
      }
      if (peek()?.type === ")") {
        index += 1;
        return;
      }
      if (!peek()) {
        report("Expected ')' to close the function call.");
        return;
      }
      report("Expected ',' or ')' after the function argument.");
      synchronize(")");
      if (peek()?.type === ",") {
        index += 1;
      } else if (peek()?.type === ")") {
        index += 1;
        return;
      }
    }
    report("Expected ')' to close the function call.");
  };
  parseExpression = () => {
    const token = peek();
    if (!token) {
      report("Expected an expression.");
      return;
    }
    if (token.type === "identifier") {
      index += 1;
      const keyword = token.value.toLowerCase();
      if ((keyword === "if" || keyword === "elseif") && peek()?.type !== "(") {
        if (!peek()) {
          report(`Expected an expression after "${token.value}".`, token);
        } else {
          parseExpression();
        }
        return;
      }
      if (keyword === "each") {
        if (peek()?.type !== "identifier") {
          report("Expected an iteration variable after 'each'.");
          return;
        }
        index += 1;
        if (peek()?.type !== "identifier" || peek().value.toLowerCase() !== "in") {
          report("Expected 'in' after the iteration variable.");
          return;
        }
        index += 1;
        parseExpression();
        return;
      }
      if (peek()?.type === "(") {
        reportUnknownFunction(token);
        index += 1;
        parseCallArguments();
      }
      parsePostfix();
      return;
    }
    if (token.type === "string" || token.type === "number") {
      index += 1;
      parsePostfix();
      return;
    }
    if (token.type === "(") {
      index += 1;
      if (peek()?.type === ")") {
        report("Expected an expression inside parentheses.");
        index += 1;
        return;
      }
      parseExpression();
      if (peek()?.type === ")") {
        index += 1;
      } else {
        report("Expected ')' to close the parenthesized expression.");
      }
      parsePostfix();
      return;
    }
    if (token.type === "invalid") {
      report(`Unexpected character "${token.value}" in Azure Pipelines expression.`, token);
    } else {
      report(`Unexpected token "${token.value}" in Azure Pipelines expression.`, token);
    }
    index += 1;
  };
  if (!tokens.length) {
    report("Expected an Azure Pipelines expression.");
    return diagnostics;
  }
  parseExpression();
  while (peek()) {
    report(`Unexpected token "${peek().value}" in Azure Pipelines expression.`);
    index += 1;
  }
  return diagnostics;
}

function getExpressionDiagnostics(text) {
  const diagnostics = [];
  const openers = [["${{", "}}"], ["$[", "]"]];
  for (const [open, close] of openers) {
    let searchFrom = 0;
    while (true) {
      const start = text.indexOf(open, searchFrom);
      if (start < 0) {
        break;
      }
      const bodyStart = start + open.length;
      const end = findExpressionEnd(text, bodyStart, close);
      if (end < 0) {
        diagnostics.push({start, end: start + open.length, message: `Unclosed Azure Pipelines expression; expected "${close}".`});
        break;
      }
      const body = text.slice(bodyStart, end);
      diagnostics.push(...parseExpressionBody(body, bodyStart));
      searchFrom = end + close.length;
    }
  }
  let yamlDocument;
  try {
    yamlDocument = YAML.parseDocument(text, {uniqueKeys: false});
  } catch {
    return diagnostics;
  }
  const visitConditions = node => {
    if (!node || typeof node !== "object") {
      return;
    }
    if (Array.isArray(node.items)) {
      node.items.forEach(visitConditions);
      return;
    }
    if (node.key?.value === "condition" && typeof node.value?.value === "string") {
      const scalar = node.value;
      if (
        !Array.isArray(scalar.range) ||
        scalar.type === "BLOCK_LITERAL" ||
        scalar.type === "BLOCK_FOLDED" ||
        /\$\{\{|\$\[|\$\(/.test(scalar.value)
      ) {
        return;
      }
      const body = scalar.value.trim();
      if (!body) {
        return;
      }
      const quoteOffset = String(scalar.type || "").startsWith("QUOTE_") ? 1 : 0;
      const leadingWhitespace = scalar.value.length - scalar.value.trimStart().length;
      diagnostics.push(...parseExpressionBody(
        body,
        scalar.range[0] + quoteOffset + leadingWhitespace
      ));
    }
    if (node.key) {
      visitConditions(node.key);
      visitConditions(node.value);
    }
  };
  visitConditions(yamlDocument.contents);
  return diagnostics;
}
